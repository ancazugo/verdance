    API reference¶

    verdance is usable as a library as well as a command line tool. Three names are importable directly from the top-level package:

    import verdance

cfg = verdance.load_config("mycity.yaml")
verdance.run(cfg, "T3", buffer=50)      # writes T3 CSVs, returns {geo_code: DataFrame | None}
merged = verdance.run(cfg, "Merge")     # consolidates and returns the merged unit table

    They are imported lazily, so import verdance — to read verdance.__version__, say — does not pull in PySpark and the rest of the stack.

    VerdanceConfig is documented alongside the YAML keys it mirrors, in the config reference.

    The per-geo-code analysis modules (T3, T30, T30_buildings, T300, Tree_count) and Merge are covered by run(). Trees, Heights, Spectral and the raster Visibility engine have bespoke setup and remain CLI-only for now.

    Public API¶

    verdance.run ¶

    Run one verdance process programmatically and return its result.

    config is a path or a loaded VerdanceConfig. process is one of T3, T30, T30_buildings, T300, Tree_count or Merge. A Spark session is created (and reused via getOrCreate) unless one is passed as sedona.

    Merge returns the merged unit DataFrame; the per-geo modules write their CSVs, record a run manifest, and return {geo_code: DataFrame | None} (None for a region that failed). For Trees, Heights, Spectral and raster Visibility, use the verdance CLI.

    def run(
    config,
    process: str,
    *,
    sedona=None,
    geo_level: str | None = None,
    sub_geo_level: str | None = None,
    geo_code: str | None = None,
    query_method: str = "sql",
    buffer: int = 100,
    tree_area: int = 10,
    tree_height: int = 3,
    low_threshold: int = 3,
    high_threshold: int = 60,
    gee_scale: float = 1.0,
    overwrite: bool = True,
    dggs: str | None = None,
    dggs_resolution: int | None = None,
    t3_buffers=(10, 25, 50, 75, 100),
    rule_t3_buffer: int = 50,
    rule_t30_buffer: int | None = None,
    rule_distance: str = "euclidean",
    rule_t3_metric: str = "proximity",
):
    """Run one verdance process programmatically and return its result.

    `config` is a path or a loaded `VerdanceConfig`. `process` is one of T3,
    T30, T30_buildings, T300, Tree_count or Merge. A Spark session is created
    (and reused via `getOrCreate`) unless one is passed as `sedona`.

    Merge returns the merged unit `DataFrame`; the per-geo modules write their
    CSVs, record a run manifest, and return `{geo_code: DataFrame | None}` (None
    for a region that failed). For Trees, Heights, Spectral and raster
    Visibility, use the `verdance` CLI.
    """
    if process != "Merge" and process not in PROCESS_FNS:
        raise NotImplementedError(
            f"verdance.run does not cover {process!r} — use the `verdance` CLI for Trees, Heights, "
            f"Spectral and raster Visibility. Programmatic processes: {sorted(PROCESS_FNS) + ['Merge']}."
        )

    cfg = config if isinstance(config, VerdanceConfig) else load_config(config)
    geo_levels = cfg.columns.geo_levels
    dggs = dggs if dggs is not None else cfg.dggs
    dggs_resolution = dggs_resolution if dggs_resolution is not None else cfg.dggs_resolution
    sedona = sedona or get_spark()

    if process == "Merge":
        from .merge import merge_output_csv, process_data

        t3b = list(t3_buffers)
        merge_output_csv(sedona, cfg, t3b, dggs=dggs, dggs_resolution=dggs_resolution)
        if dggs is not None:
            merge_geo = geo_level or sub_geo_level or get_system(dggs).column_name(dggs_resolution)
            merge_sub = sub_geo_level or get_system(dggs).column_name(dggs_resolution)
        else:
            merge_geo = geo_level or (geo_levels[-2] if len(geo_levels) > 1 else geo_levels[0])
            merge_sub = sub_geo_level or geo_levels[-1]
        return process_data(
            sedona, cfg, merge_geo, merge_sub, t3b, dggs=dggs, dggs_resolution=dggs_resolution,
            rule_t3_buffer=rule_t3_buffer, rule_t30_buffer=rule_t30_buffer,
            rule_distance=rule_distance, rule_t3_metric=rule_t3_metric,
        )

    # only T300 walks the road network and parks; the rest skip ~900 MB of reads
    tables = load_tables(sedona, cfg, dggs=dggs, dggs_resolution=dggs_resolution,
                         load_roads_parks=(process == "T300"))
    dirs = tables["output_dirs"]
    gl = geo_level or geo_levels[0]
    sgl = sub_geo_level or (geo_levels[-1] if len(geo_levels) > 1 else geo_levels[0])
    output_dir = dirs[_OUTPUT_DIR_KEY[process]]
    if dggs is not None:
        sgl = get_system(dggs).column_name(dggs_resolution)
        if process in ("T30", "Tree_count"):
            # grid runs sit beside the census-unit outputs instead of overwriting them
            output_dir = dirs["base"] / unit_output_name(process, dggs, dggs_resolution)
            output_dir.mkdir(parents=True, exist_ok=True)

    args_dict = {
        "sedona": sedona, "geo_level": gl, "sub_geo_level": sgl, "cfg": cfg, "output_dir": output_dir,
        "query_method": query_method, "buffer": buffer, "tree_area": tree_area, "tree_height": tree_height,
        "low_threshold": low_threshold, "high_threshold": high_threshold, "gee_scale": gee_scale,
        "overwrite": overwrite, "road_nodes_gdf": tables.get("road_nodes_gdf"),
        "road_edges_gdf": tables.get("road_edges_gdf"),
        # already filtered by load_tables; passing them keeps T300 from re-reading
        # and re-filtering the parks datasets once per geo code
        "parks_sites_gdf": tables.get("parks_sites_gdf"),
        "parks_access_gdf": tables.get("parks_access_gdf"),
    }
    codes = list([geo_code] if geo_code else tables["census_boundaries_gdf"][gl].unique())
    start = time.time()
    results, failed = {}, []
    for code in codes:
        try:
            result = run_process_code(process, args_dict, code)
        except Exception:
            logger.exception(f"{process}: error processing {code}")
            result = None
        results[code] = result
        if result is None:
            failed.append(code)
    write_run_manifest(
        output_dir, process, codes, [c for c in codes if c not in failed], failed,
        cfg=cfg, duration_s=time.time() - start,
        extra={"geo_level": gl, "sub_geo_level": sgl, "buffer": buffer},
    )
    return results

    def load_config(path: str | Path) -> VerdanceConfig:
    """Load and validate a verdance YAML config into a `VerdanceConfig`.

    Unknown keys, missing required sections and a geographic `crs` are rejected
    here rather than part-way through a long Spark run.
    """
    with open(path) as f:
        raw = yaml.safe_load(f)

    if not isinstance(raw, dict):
        kind = "empty" if raw is None else f"a {type(raw).__name__}"
        raise ValueError(f"Config file {path} is {kind}; expected a YAML mapping of settings")

    _require(raw, "study_area_name", "crs", "data", "columns", "output")
    _reject_unknown(raw, {f.name for f in fields(VerdanceConfig)}, "top-level config")

    data_raw = raw["data"]
    _require_section(data_raw, "data", "buildings", "parks_sites", "parks_access", "roads", "census_boundaries")
    _reject_unknown(data_raw, {f.name for f in fields(DataPaths)}, "data")

    columns_raw = raw["columns"]
    _require_section(columns_raw, "columns", "geo_levels")
    _reject_unknown(columns_raw, {f.name for f in fields(ColumnMapping)}, "columns")
    if not columns_raw.get("geo_levels"):
        raise ValueError("columns.geo_levels must be a non-empty list")
    if building_source(data_raw["buildings"]) is None and not columns_raw.get("building_id"):
        raise ValueError("columns.building_id is required when data.buildings is a file path")

    _validate_osm_sources(data_raw, columns_raw)
    osm_cfg = _parse_osm_section(raw.get("osm") or {})
    open_buildings_cfg = _parse_open_buildings_section(raw.get("open_buildings") or {})
    overture_cfg = _parse_overture_section(raw.get("overture") or {})
    tree_seg_cfg = _parse_tree_segmentation_section(raw.get("tree_segmentation") or {})
    heights_cfg = _parse_heights_section(raw.get("heights") or {})
    visibility_cfg = _parse_visibility_section(raw.get("visibility") or {})
    terrain_cfg = _parse_terrain_section(raw["terrain"] if "terrain" in raw else {})
    context_buffer = raw.get("context_buffer", 100.0)
    if isinstance(context_buffer, bool) or not isinstance(context_buffer, (int, float)) or context_buffer < 0:
        raise ValueError(f"context_buffer must be a non-negative number (metres), got {context_buffer!r}")

    output_raw = raw["output"]
    _require_section(output_raw, "output", "base_dir")
    _reject_unknown(output_raw, {f.name for f in fields(OutputPaths)}, "output")

    tile_raw = raw.get("tile_system", {})

    try:
        from pyproj import CRS
        crs_obj = CRS.from_user_input(raw["crs"])
    except Exception as e:
        raise ValueError(f"Invalid CRS '{raw['crs']}': {e}")
    if crs_obj.is_geographic:
        raise ValueError(
            f"crs '{raw['crs']}' is geographic (degrees). verdance measures areas in m² and buffers in "
            "metres, so a projected/metric CRS is required — e.g. the local UTM zone or a national grid."
        )

    dggs, dggs_resolution = _parse_dggs(raw)

    return VerdanceConfig(
        study_area_name=raw["study_area_name"],
        crs=raw["crs"],
        data=DataPaths(
            buildings=data_raw["buildings"],
            parks_sites=data_raw["parks_sites"],
            parks_access=data_raw["parks_access"],
            roads=data_raw["roads"],
            census_boundaries=data_raw["census_boundaries"],
            road_nodes=data_raw.get("road_nodes"),
            trees_dir=data_raw.get("trees_dir"),
            chm_tiles_dir=data_raw.get("chm_tiles_dir"),
            chm_pattern=data_raw.get("chm_pattern") or "*.tif",
            chm_overlap=_parse_chm_overlap(data_raw.get("chm_overlap")),
            chm_cache_dir=data_raw.get("chm_cache_dir"),
            canopy_height_ee_path=data_raw.get("canopy_height_ee_path"),
            meta_chm=_parse_meta_chm(data_raw.get("meta_chm")),
        ),
        columns=ColumnMapping(
            building_id=columns_raw.get("building_id"),
            building_layer=columns_raw.get("building_layer"),
            building_height_col=columns_raw.get("building_height_col"),
            building_use_col=columns_raw.get("building_use_col"),
            building_use_value=columns_raw.get("building_use_value"),
            road_node_id=columns_raw.get("road_node_id", "id"),
            road_edge_start=columns_raw.get("road_edge_start", "start_node"),
            road_edge_end=columns_raw.get("road_edge_end", "end_node"),
            road_edge_length=columns_raw.get("road_edge_length", "length"),
            road_edge_layer=columns_raw.get("road_edge_layer", "road_link"),
            road_node_layer=columns_raw.get("road_node_layer", "road_node"),
            park_id=columns_raw.get("park_id", "id"),
            park_function_col=columns_raw.get("park_function_col"),
            park_function_value=columns_raw.get("park_function_value"),
            park_access_ref_col=columns_raw.get("park_access_ref_col"),
            tree_height_col=columns_raw.get("tree_height_col", "height"),
            tree_area_col=columns_raw.get("tree_area_col", "area"),
            tree_id_col=columns_raw.get("tree_id_col", "treeID"),
            tree_layer=columns_raw.get("tree_layer", "trees"),
            geo_levels=columns_raw["geo_levels"],
            geo_level_labels=_level_mapping(columns_raw, "geo_level_labels"),
            geo_level_names=_level_mapping(columns_raw, "geo_level_names"),
        ),
        output=OutputPaths(base_dir=output_raw["base_dir"]),
        gee_project=raw.get("gee_project"),
        gee_boundaries_asset=raw.get("gee_boundaries_asset"),
        park_min_area_ha=_parse_park_min_area(raw.get("park_min_area_ha")),
        dggs=dggs,
        dggs_resolution=dggs_resolution,
        tile_system=TileSystemConfig(
            enabled=tile_raw.get("enabled", False),
            tile_name_pattern=tile_raw.get("tile_name_pattern"),
        ),
        osm=osm_cfg,
        overture=overture_cfg,
        open_buildings=open_buildings_cfg,
        tree_segmentation=tree_seg_cfg,
        heights=heights_cfg,
        visibility=visibility_cfg,
        terrain=terrain_cfg,
        context_buffer=float(context_buffer),
    )
